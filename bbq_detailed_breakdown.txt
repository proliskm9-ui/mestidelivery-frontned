def blend_onto_plate(food_bgr, alpha_mask, offset_x=0, offset_y=0, scale=1.0, shadow_strength=0.65):
    """Composites segmented food onto master_bg with realistic AO + Directional Cast Shadow."""
    h, w = food_bgr.shape[:2]
    if scale != 1.0:
        food_bgr = cv2.resize(food_bgr, (0, 0), fx=scale, fy=scale, interpolation=cv2.INTER_LANCZOS4)
        alpha_mask = cv2.resize(alpha_mask, (0, 0), fx=scale, fy=scale, interpolation=cv2.INTER_LANCZOS4)
    
    fh, fw = food_bgr.shape[:2]
    
    # Target canvas
    full_food = np.zeros((1024, 1024, 3), dtype=np.float32)
    full_alpha = np.zeros((1024, 1024), dtype=np.float32)
    
    # Center placement
    start_x = int(plate_cx - fw // 2 + offset_x)
    start_y = int(plate_cy - fh // 2 + offset_y)
    
    # Clip bounds
    src_x1 = max(0, -start_x)
    src_y1 = max(0, -start_y)
    src_x2 = min(fw, 1024 - start_x)
    src_y2 = min(fh, 1024 - start_y)
    
    dst_x1 = max(0, start_x)
    dst_y1 = max(0, start_y)
    dst_x2 = dst_x1 + (src_x2 - src_x1)
    dst_y2 = dst_y1 + (src_y2 - src_y1)
    
    if dst_x2 <= dst_x1 or dst_y2 <= dst_y1:
        return master_bg
        
    full_food[dst_y1:dst_y2, dst_x1:dst_x2] = food_bgr[src_y1:src_y2, src_x1:src_x2]
    full_alpha[dst_y1:dst_y2, dst_x1:dst_x2] = alpha_mask[src_y1:src_y2, src_x1:src_x2]
    
    # 1. Ambient Occlusion (tight contact crease under food)
    ao_kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (9, 9))
    ao_dilated = cv2.dilate(full_alpha, ao_kernel, iterations=1)
    ao_blur = cv2.GaussianBlur(ao_dilated, (9, 9), 2.5)
    ao_vis = np.clip(ao_blur - full_alpha, 0.0, 1.0)
    ao_factor = 1.0 - (ao_vis * shadow_strength * 0.7)
    
    # 2. Directional Cast Shadow (+6px X, +10px Y towards bottom-right)
    cast_kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (25, 25))
    cast_dilated = cv2.dilate(full_alpha, cast_kernel, iterations=1)
    M_cast = np.float32([[1, 0, 8], [0, 1, 14]])
    cast_shifted = cv2.warpAffine(cast_dilated, M_cast, (1024, 1024))
    cast_blur = cv2.GaussianBlur(cast_shifted, (35, 35), 9.0)
    cast_vis = np.clip(cast_blur - full_alpha, 0.0, 1.0)
    cast_factor = 1.0 - (cast_vis * shadow_strength * 0.5)
    
    # Combined plate shadow
    total_shadow = np.minimum(ao_factor, cast_factor)[:, :, None]
    
    bg_shadowed = master_bg.astype(np.float32) * total_shadow
    
    # Alpha compositing
    alpha_3d = np.dstack([full_alpha, full_alpha, full_alpha])
    comp = full_food * alpha_3d + bg_shadowed * (1.0 - alpha_3d)
    
    # Final subtle contrast
    gauss = cv2.GaussianBlur(comp, (0, 0), 1.0)
    final = np.clip(cv2.addWeighted(comp, 1.08, gauss, -0.08, 0), 0, 255).astype(np.uint8)
    return final

def grade_to_burger(img_bgr):
    if img_bgr.shape[:2] != (900, 900):
        img_bgr = cv2.resize(img_bgr, (900, 900), interpolation=cv2.INTER_LANCZOS4)
        
    # 1. Gentle gamma lift on shadows & midtones (1.08)
    inv_gamma = 1.0 / 1.08
    table = np.array([((i / 255.0) ** inv_gamma) * 255 for i in np.arange(0, 256)]).astype("uint8")
    graded = cv2.LUT(img_bgr, table)
    
    # 2. HSV grading: boost warmth (channel 1 - saturation) and brightness (channel 2 - value)
    hsv = cv2.cvtColor(graded, cv2.COLOR_BGR2HSV).astype(np.float32)
    hsv[:, :, 2] = np.clip(hsv[:, :, 2] * 1.10, 0, 255) # +10% brightness
    hsv[:, :, 1] = np.clip(hsv[:, :, 1] * 1.08, 0, 255) # +8% warm saturation
    bgr = cv2.cvtColor(hsv.astype(np.uint8), cv2.COLOR_HSV2BGR).astype(np.float32)
    
    # 3. Add warm golden cast to plate & table (lift red & green channels slightly by +4 to +6)
    bgr[:, :, 2] = np.clip(bgr[:, :, 2] * 1.04 + 3, 0, 255) # Red
    bgr[:, :, 1] = np.clip(bgr[:, :, 1] * 1.02 + 2, 0, 255) # Green
    
    # 4. Micro-contrast
    gauss = cv2.GaussianBlur(bgr, (0, 0), 1.0)
    final = np.clip(cv2.addWeighted(bgr, 1.08, gauss, -0.08, 0), 0, 255).astype(np.uint8)
    return final
